WAF service configuration (docker-compose.yml)
services:
  web:
    image: bkimminich/juice-shop
    ports:
      - "3000:3000"

  waf:
    image: owasp/modsecurity-crs:nginx
    ports:
      - "8080:8080"
    environment:
      BACKEND: "http://web:3000"
      MODSEC_RULE_ENGINE: "On"
      BLOCKING_PARANOIA: "1"

Active Nginx configuration excerpts
proxy_pass http://web:3000;
modsecurity on;
modsecurity_rules_file /etc/modsecurity.d/setup.conf;
